setup
{
	//scale 0.015444		// Set default scale to 16/30.5 since world is in 16 units per foot and model is in cm's
	scale 0.50			// Scaled down a bit to fit in a player's hand
	//path models/weapons/M2FragGrenade

	//surface grenade shader M2FragGrenade
	/*

	path models/snowcontest
	skelmodel snowball.skd
	surface snowball shader snowball
}

init
{
	server
	{
		classname		Weapon
		weapontype		grenade
		name			"Frag Grenade"
		rank			510 510

		pickupsound		m2frag_snd_pickup
		ammopickupsound	m2frag_snd_pickup_ammo
		noammosound		m2frag_snd_noammo

		// Holstering info
//		holstertag		"Bip01 Spine1"
//		holsteroffset	"3.12 10.5 3.64"
//		holsterangles	"20 -40 0"
//		holsterScale	1.0

		// Primary fire type info (fast, arcing throw)
		firetype		projectile
		ammotype		"grenade"
		projectile		models/static/rock_winter_medium.tik
		semiauto
		clipsize		1
		startammo		1
		ammorequired	0
		firedelay		0.75
		maxchargetime	4.0
		minchargetime	0.6
		crosshair		0
		quiet // don't notify AI of it being fired
		autoputaway 	1 // automatically put the weapon away when out of ammo
		usenoammo		0 // don't allow it to be used when it has no ammo
		movementspeed	1.0

		// Secondary fire type info (more controllable, straight throw)
		shareclip
		secondary firetype		projectile
		secondary ammotype		"grenade"
		secondary projectile	models/static/rock_winter_medium.tik
		secondary clipsize		1
		secondary ammorequired	0
		secondary firedelay		0.75
		secondary maxchargetime	1.5
		secondary minchargetime	0.5

		// AI animation group info
		weapongroup		grenade

		// DM Attributes
		dmprojectile		models/static/rock_winter_medium.tik
		dmstartammo			6
		dmammorequired		1
		//dmusenoammo			1
		dmfiredelay			0.75
		dmcrosshair			0
		dmmovementspeed 	1.0
		secondary dmprojectile	models/static/rock_winter_medium.tik
		secondary dmammorequired	1
		secondary dmfiredelay		0.75

		cache models/static/rock_winter_medium.tik
		cache models/static/rock_winter_medium.tik
	}
	client
	{
	}
}

animations
{
	raise snowball.skc
	{
		// if we're pulling out grenades, we've definatly got 'em
		server
		{
			enter surface snowball -nodraw
		}
	}
	idle snowball.skc
	{
		server
		{
			enter surface snowball -nodraw
		}
	}

	// this should make sure that it doesn't look like there's a grenade in hand when you don't have one
	idle_empty snowball.skc
	{
		server
		{
			enter surface snowball +nodraw
		}
	}

	charge snowball.skc
	{
		client
		{
			enter sound snd_step_snow weapon
		}
	}
	secondarycharge snowball.skc
	{
		client
		{
			enter sound snd_step_snow weapon
		}
	}
	
//	fire grenade_fire.skc
	fire snowball.skc
	{
		server
		{
			first shoot
			first surface snowball +nodraw
//			15 surface snowball -nodraw
		}
		client
		{
			first sound snd_step_snow weapon
		}
	}
	secondaryfire snowball.skc
	{
		server
		{
			first shoot secondary
			first surface snowball +nodraw
//			15 surface snowball -nodraw
		}
		client
		{
			first sound snd_step_snow weapon
		}
	}

	reload snowball.skc
	{
		server
		{
			first surface snowball -nodraw
		}
	}
}
